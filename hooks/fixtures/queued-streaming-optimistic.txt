  const sendStreamingPrompt = useCallback(async (
    message: string,
    behavior: "steer" | "followUp",
    images?: AttachedImage[],
  ) => {
    const sid = sessionIdRef.current;
    const restore = () => restoreSubmission(message, images, composerDraftKey);
    if (sessionStatePendingRef.current || editEntryId) {
      restore();
      return;
    }
    if (!sid) {
      restore();
      addNotice({ type: "error", message: "No active session for the queued message" });
      return;
    }
    cancelInitialLanding();
    const piImages = images?.map((img) => ({ type: "image" as const, data: img.data, mimeType: img.mimeType }));
    const optimisticUserMessage: AgentMessage = {
      role: "user",
      content: piImages?.length
        ? [
            ...(message.trim() ? [{ type: "text" as const, text: message }] : []),
            ...piImages.map((img) => ({
              type: "image" as const,
              source: { type: "base64" as const, media_type: img.mimeType, data: img.data },
            })),
          ]
        : message,
      timestamp: Date.now(),
    };
    const optimisticKey = userMessageKey(optimisticUserMessage);
    const baselineCount = messagesRef.current.filter((item) => userMessageKey(item) === optimisticKey).length
      + pendingQueuedUserMessagesRef.current.filter((item) => item.key === optimisticKey).length;
    pendingQueuedUserMessagesRef.current.push({ message: optimisticUserMessage, key: optimisticKey, baselineCount });
    messagesRef.current = [...messagesRef.current, optimisticUserMessage];
    setMessages((prev) => [...prev, optimisticUserMessage]);
    pendingScrollToUserRef.current = true;
    setPromptAnchorActive(true);

    const queueKind = behavior === "steer" ? "steering" : "followUp";
    rememberQueuedDraftMessage(queuedDraftMessagesRef.current, queueKind, message, images);
    if (sid) setQueuedDrafts(sid, queuedDraftMessagesRef.current);
    try {
      await sendAgentCommandR(sid, {
        type: "prompt",
        message,
        streamingBehavior: behavior,
        ...(piImages?.length ? { images: piImages } : {}),
      });
    } catch (e) {
      console.error("Failed to submit streaming prompt:", e);
      // A transport failure after dispatch is ambiguous: the server may have
      // accepted the queued prompt before the response was lost. Restoring in
      // that case would invite a duplicate turn.
      forgetQueuedDraftMessage(queuedDraftMessagesRef.current, queueKind, message);
      if (sid) setQueuedDrafts(sid, queuedDraftMessagesRef.current);
      if (isPromptRejectedError(e)) {
        pendingQueuedUserMessagesRef.current = pendingQueuedUserMessagesRef.current.filter((item) => item.message !== optimisticUserMessage);
        messagesRef.current = messagesRef.current.filter((item) => item !== optimisticUserMessage);
        setMessages((prev) => prev.filter((item) => item !== optimisticUserMessage));
        restore();
      }
      addNotice({
        type: "error",
        message: e instanceof Error ? e.message : String(e),
      });
    }
  }, [addNotice, cancelInitialLanding, composerDraftKey, restoreSubmission, sendAgentCommandR, editEntryId]);

  const handleSteer = useCallback(async (message: string, images?: AttachedImage[]) => {
    await sendStreamingPrompt(message, "steer", images);
  }, [sendStreamingPrompt]);
